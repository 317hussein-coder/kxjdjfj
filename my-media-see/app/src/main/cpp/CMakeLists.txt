cmake_minimum_required(VERSION 3.22.1)
project(my_media_see LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(NATIVE_APP_GLUE ${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c)
add_library(main SHARED src/main/cpp/main.cpp ${NATIVE_APP_GLUE})
target_include_directories(main PRIVATE ${ANDROID_NDK}/sources/android/native_app_glue)
target_compile_options(main PRIVATE -Wall -Wextra -Werror=return-type)
target_link_libraries(main PRIVATE android log mediandk)
